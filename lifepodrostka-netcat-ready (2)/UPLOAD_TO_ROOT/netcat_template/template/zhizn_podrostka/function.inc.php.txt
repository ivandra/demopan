<?php
function get_utm(){ $values=array();foreach(array('utm_source','utm_medium','utm_campaign','utm_content','utm_term','yclid','gclid') as $k){if(isset($_GET[$k])&&is_string($_GET[$k])){$values[$k]=mb_substr($_GET[$k],0,250);}}if($values){$_SESSION['utm_get']=http_build_query($values);}return (string)($_SESSION['utm_get']??'');}
function shortcode($text){return preg_replace_callback('/\[([a-zA-Z_][a-zA-Z0-9_]*)\]/',function($m){return (string)nc_core()->widget->generate('regions',array($m[1]));},(string)$text);}
function shortcode_component($text){return shortcode($text);}
function formatDate($day,$month,$year){$d=DateTime::createFromFormat('!d-m-Y',sprintf('%02d-%02d-%04d',$day,$month,$year));return $d?$d->format('d.m.Y'):'';}
